Hefesto LCE (Publicación libros)

Proceso encargado de enviar un Libro Contable Electrónico (LCE) al Servicio de Impuestos Internos (SII). Hefesto utiliza el archivo XML generado, configura el ambiente de publicación y los datos de autenticación, y realiza el envío directamente al SII. Una vez procesada la solicitud, el sistema informa el resultado de la operación y, cuando la publicación es exitosa, obtiene el **Track ID** entregado por el SII, permitiendo identificar y realizar el seguimiento del envío.


Ejemplo de publicación de libros en el SII.



/// <summary>
/// Bibliotecas necsarias.
/// </summary>
using Hefesto.SII.Lce.Enumeradores;
using Hefesto.SII.Lce.Extruturas.Balance;
using Hefesto.SII.Lce.Extruturas.Diario;
using Hefesto.SII.Lce.Extruturas.Diccionario;
using Hefesto.SII.Lce.Extruturas.Mayor;
using Hefesto.SII.Lce.Instancias;
using System.Text;
using System.Xml.Linq;            

internal class Program
{

    ///
    /// Punto de entrada del programa.
    static async Task Main(string[] args)
    {
        ///
        /// Publicación de libros en el SII.
        await PublicarEnvioLibroSII();

        ///
        /// Mantener la consola abierta.
        Console.readLine();

    }

     /// 
    /// Publica ene l SII un libro generado.
    /// 
    public static async Task PublicarEnvioLibroSII()
    {

        ///
        /// fullpath del Libro a publicar.
        var libro = "EnvioLibroDiario.xml";

        ///
        /// Cree el publicador 
        var publicador = HefestoLceElements.CrearSiiLcePublicador()
        
            .SiiConfiguracion( cfg => {

                ///
                /// Datos para la autenticación con el sii.
                cfg.SiiAutenticacion(auth =>
                {
                    auth.Ambiente = SiiAmbiente.CERTIFICACION;
                    auth.Basse64 = "MIIPNgIBAzCCDvIGCSqGSIb3DQEHAaCCDuMEgg7fMIIO2zCCBgwGCSqGSIb3DQE==";
                    auth.Password = "1234";
                });

            });

        ///
        /// Publicar el libro.
        var respPub = 
            await 
                publicador.
                    Libro(libro).
                        SIIPublicarLibro();


        ///
        /// Resultado en consola.
        Console.WriteLine($"EsCorrceto : {respPub.Success}");
        Console.WriteLine($"Mensaje    : {respPub.Message}");


        ///
        /// Si la publicación fue exitosa, mostrar el contenido de la respuesta.
        /// respPub.Data contiene el contenido de la respuesta del SII trackid.
        if (respPub.Success)
            Console.WriteLine($"Data: {respPub.Data}");

    }
}
