Hefesto LCE (Libro Diario Obligatorio)

El Diccionario de Cuentas es el Libro Contable Electrónico utilizado para establecer la correspondencia entre las cuentas contables del contribuyente y las cuentas definidas en el Plan de Cuentas del SII. Esta homologación permite representar la estructura contable de la empresa bajo la clasificación requerida por el Servicio, sin modificar el plan de cuentas utilizado internamente.
En Hefesto SII LCE, el Diccionario se construye mediante estructuras que facilitan su generación y posterior incorporación al Envío Obligatorio, junto con el Balance y los demás comprobantes requeridos.

/// Bibliotecas necsarias.
/// </summary>

        GenerarEnviolibroDiarioOblitatorio();
    }

	/// <summary>
	/// Crear el envio de libro diario obligatorio.
	/// </summary>
	public static void GenerarEnviolibroDiccionarioOblitatorio()
	{
		///
		/// crear el libro SiiLceEnvioLibros
		var bookDiarioOblig = HefestoLceElements.CrearSiiLceEnvioOblig()

			///
			/// Configuracion del libro
			.SiiConfiguracion(cfg =>
			{

				///
				/// Datos para la firma del documento.
				cfg.SiiAutenticacion(auth =>
				{
					auth.Basse64 = "MIIPNgIBAzCCDvIGCSqGSIb3DQEHAaCCDuMEgg7fMIIO2zCCBgwGCSq==";
					auth.Password = "1234";
				});

				///
				/// Datos del certificado del SII.
				cfg.SIILceCoCertificado(auth =>
				{
					auth.Load = "CAL_99999999-9_C_2_1.xml";
				});

			})


			///
			/// Documento envio obligatorio.
			.SiiDocumentoEnvioOblig(doc =>
			{

				doc.SiiCaratula(car =>
				{
					car.RutEnvia = "88888888-8";
					car.RutContribuyente = "99999999-9";
				});

				doc.SIILceCoCierre(cierre =>
				{
					///
					/// agregar libro diario res.
					cierre.SIIDiarioRes( GenerarLibroDiarioRes() );
				});

			});

		///
		/// Despliega en consola el resultado de la exportación del libro.
		var respExporta = bookDiarioOblig.ExportarLibro();
		Console.WriteLine($"EsCorrceto : {respExporta.Success}");
		Console.WriteLine($"Mensaje    : {respExporta.Message}");

		///
		/// Si es correcto, guarde el libro en un archivo XML.
		if (respExporta.Success)
    		File.WriteAllText(
				"EnviosObligatorios\\EnvioLibroDiarioObligatorio.xml",
					respExporta.Data,
						Encoding.GetEncoding("ISO-8859-1"));

		///
		/// Generación del libro diario res
		HefLceDiarioRes GenerarLibroDiarioRes()
		{

			///
			/// Cree el libro.
			var libro = HefestoLceElements.CrearSiiLibroDiarioRes();

			///
			/// Complete el libro.
			libro.SiiDocumentoDiarioRes(res => {

				res.SiiIdentificacion(ident => {

					ident.RutContribuyente = "99999999-8";

					ident.SiiPeriodoTributario(periodo => {

						periodo.Inicial = "2026-01";
						periodo.Final = "2026-01";

					});

				});

				res.RutFirma = "88888888-8";

				res.SiiRegistrosDiario(GenerarRegistrosDiarios());

			});

			///
			/// Regrese el libro.
			return libro;

		}

		///
		/// Crea la lista de los registros diarios del libro Diario
		List GenerarRegistrosDiarios()
		{
			///
			/// Lista de comprobantes diarios
			var registrosDiarios = new List();

			///
			/// Registro #1
			var registro = new HefRegistroDiario
			{

			registrosDiarios.Add(registro);

			///
