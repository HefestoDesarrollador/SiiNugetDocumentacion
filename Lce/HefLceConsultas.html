Hefesto LCE (Consultas libros)

Proceso encargado de enviar un Libro Contable Electrónico (LCE) al Servicio de Impuestos Internos (SII). Hefesto utiliza el archivo XML generado, configura el ambiente de publicación y los datos de autenticación, y realiza el envío directamente al SII. Una vez procesada la solicitud, el sistema informa el resultado de la operación y, cuando la publicación es exitosa, obtiene el **Track ID** entregado por el SII, permitiendo identificar y realizar el seguimiento del envío.


Ejemplo de consulta estado unificado de libros en el SII.



/// <summary>
/// Bibliotecas necsarias.
/// </summary>
using Hefesto.SII.Lce.Enumeradores;
using Hefesto.SII.Lce.Extruturas.Balance;
using Hefesto.SII.Lce.Extruturas.Diario;
using Hefesto.SII.Lce.Extruturas.Diccionario;
using Hefesto.SII.Lce.Extruturas.Mayor;
using Hefesto.SII.Lce.Instancias;
using System.Text;
using System.Xml.Linq;            

internal class Program
{

    ///
    /// Punto de entrada del programa.
    static async Task Main(string[] args)
    {
        ///
        /// Publicación de libros en el SII.
        await PublicarEnvioLibroSII();

        ///
        /// Mantener la consola abierta.
        Console.readLine();

    }

    ///
    /// Consultas al SII por el estado de los libros enviados.
    public static async Task ConsultarEstadoEnvioLibroSII()
    {

        ///
        /// Regresa el estado de todos los envíos al SII.

        ///
        /// Cree el publicador 
        var consultas = HefestoLceElements.CrearSiiLceConsultas()

            .SiiConfiguracion(cfg => {

                ///
                /// Datos para la autenticación con el sii.
                cfg.SiiAutenticacion(auth =>
                {
                    auth.Ambiente = SiiAmbiente.PRODUCCION;
                    auth.Basse64 = "MIIPNgIBAzCCDvIGCSqGSIb3DQEHAaCCDuMEgg7fMIIO2zCCB";
                    auth.Password = "1234";
                });

            });

        ///
        /// Consultar el estado del envío utilizando el trackid.
        var respConsulta =
            await
                consultas.
                    RutEmpresa("99999999-9").
                        Periodo(2024).
                            SIIConsultarEstadoEnvioLibro();
                    

        ///
        /// Display.
        Console.WriteLine($"EsCorrceto : {respConsulta.Success}");
        Console.WriteLine($"Mensaje    : {respConsulta.Message}");
        if (respConsulta.Success)
        {

            File.WriteAllText(
                "Consultas\\ConsultaEstadosLibros_Produccion.json", 
                respConsulta.Data, 
                Encoding.GetEncoding("ISO-8859-1"));
        }

    }
    
}
