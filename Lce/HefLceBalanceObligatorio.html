Hefesto LCE (Libro Balance Obligatorio)

El Balance Obligatorio es un Libro Contable Electrónico (LCE) utilizado para informar al Servicio de Impuestos Internos (SII) la información contable correspondiente a un período tributario.
Desde el punto de vista del desarrollo, el LCE debe generarse respetando la estructura y las reglas de validación establecidas por el SII. La información se organiza a partir de las cuentas contables utilizadas por el contribuyente, considerando sus saldos y movimientos para representar de forma consistente la situación contable del período informado.
En Hefesto SII LCE, este proceso se abstrae mediante objetos y métodos destinados a facilitar la construcción, estructuración y posterior publicación del libro, reduciendo la complejidad asociada a la generación del XML requerido por el SII.
El resultado final es un documento XML estructurado conforme al formato definido para el LCE, listo para ser validado y utilizado en el proceso de envío correspondiente.


Ejemplo de generación libro balance obligatorio


/// <summary>
/// Bibliotecas necsarias.
/// </summary>
using Hefesto.SII.Lce.Enumeradores;
using Hefesto.SII.Lce.Extruturas.Balance;
using Hefesto.SII.Lce.Extruturas.Diario;
using Hefesto.SII.Lce.Extruturas.Diccionario;
using Hefesto.SII.Lce.Extruturas.Mayor;
using Hefesto.SII.Lce.Instancias;
using System.Text;
using System.Xml.Linq;            

internal class Program
{
    ///
    /// Punto de entrada del programa.
    static async Task Main(string[] args)
    {
        ///
        /// Generar el libro balance obligatorio.
        GenerarEnviolibroBalanceOblitatorio();
    }

    /// 
    /// Crear el envio de libro balance obligatorio.
    /// 
    public static void GenerarEnviolibroBalanceOblitatorio()
    {
        
        ///
        /// crear el libro SiiLceEnvioLibros
        var bookBalanceOblig = HefestoLceElements.CrearSiiLceEnvioOblig()

            ///
            /// Configuracion del libro
            .SiiConfiguracion(cfg =>
            {
                ///
                /// Datos para la firma del documento.
                cfg.SiiAutenticacion(auth =>
                {
                    auth.Basse64 = "MIIPGQIBAzCCDtkGCSqGSIb3="; /// Certificado en base64.
                    auth.Password = "123456789";                /// Password del certificado.
                });

                ///
                /// Datos del certificado.
                cfg.SIILceCoCertificado(auth =>
                {
                    auth.Load = "{Ruta}\\CAL_999999999-9_C_2_1.xml"; /// Ruta del certificado del SII.
                });

            })


            ///
            /// Documento envio obligatorio.
            .SiiDocumentoEnvioOblig(doc =>
            {
                doc.SiiCaratula(car =>
                {
                    car.RutEnvia = "88888888-8";                          /// Rut del contribuyente que envia el libro.
                    car.RutContribuyente = "999999999-9";                 /// Rut del contribuyente que genera el libro.
                });

                doc.SIILceCoCierre(cierre =>
                {
                    cierre.SIIBalanceRes(GenerarEnvioLibroBalance());     /// Agregar libro balance obligatorio.
                });

            });

        ///
        /// Despliega en consola el resultado de la exportación del libro.
        var respExporta = bookBalanceOblig.ExportarLibro();
        Console.WriteLine($"EsCorrceto : {respExporta.Success}");
        Console.WriteLine($"Mensaje    : {respExporta.Message}");

        ///
        /// Si es correcto, guarde el libro en un archivo XML.
        if (respExporta.Success)
            File.WriteAllText(
                "EnviosObligatorios\\EnvioLibroBalanceObligatorio.xml",
                    respExporta.Data,
                        Encoding.GetEncoding("ISO-8859-1"));
                        
        
        /* Complementos del libro */                
                        

        ///
        /// Crear el libro diccionario.
        HefLceBalance GenerarEnvioLibroBalance()
        {
            ///
            /// Cree el libro.
            var libro = HefestoLceElements.CrearSiiLibroBalance();

            libro.SiiDocumentoBalance(doc => {


                doc.SiiIdentificacion(ident => {
                    ident.RutContribuyente = "999999999-9"; /// Rut del contribuyente que genera el libro.
                    ident.PeriodoTributario = "2026-07";    /// Periodo tributario del libro.
                    ident.Moneda = "CLP";                   /// Moneda del libro.
                    ident.Rectificatoria = 1;               /// Indica si el libro es rectificatorio (1) o no (0).
                });

                doc.RutFirma = "88888888-8";                /// Rut del contribuyente que firma el libro.
                doc.SiiCuentas(GenerarCuentasBalance());    /// Agregar las cuentas del libro balance.

            });

            ///
            /// Regrese el libro diccionario.
            return libro;

        }

        ///
        /// Crea las cuentas del balance.
        /// En este ejemplo, se crean dos cuentas de ejemplo.
        List GenerarCuentasBalance()
        {

            List cuentas = new();

            var cuenta = new HefCuentaBalance();
            cuenta.CodigoCuenta = "110101";
            cuenta.Debe = 0;
            cuenta.Haber = 1200;
            cuenta.SaldoDeudor = 0;
            cuenta.SaldoAcreedor = 1200;

            cuentas.Add(cuenta);

            cuenta = new HefCuentaBalance();
            cuenta.CodigoCuenta = "110140";
            cuenta.Debe = 2370513;
            cuenta.Haber = 500;
            cuenta.SaldoDeudor = 500;
            cuenta.SaldoAcreedor = 0;

            cuentas.Add(cuenta);

            ///
            /// Regrese las cuentas.
            return cuentas;
        }

    }

}
