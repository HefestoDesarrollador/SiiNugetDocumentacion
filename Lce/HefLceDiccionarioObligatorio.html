Hefesto LCE (Libro Diccionario Obligatorio)

El Diccionario de Cuentas es el Libro Contable Electrónico utilizado para establecer la correspondencia entre las cuentas contables del contribuyente y las cuentas definidas en el Plan de Cuentas del SII. Esta homologación permite representar la estructura contable de la empresa bajo la clasificación requerida por el Servicio, sin modificar el plan de cuentas utilizado internamente.
En Hefesto SII LCE, el Diccionario se construye mediante estructuras que facilitan su generación y posterior incorporación al Envío Obligatorio, junto con el Balance y los demás comprobantes requeridos.


Ejemplo de generación libro diccionario obligatorio.



/// <summary>
/// Bibliotecas necsarias.
/// </summary>
using Hefesto.SII.Lce.Enumeradores;
using Hefesto.SII.Lce.Extruturas.Balance;
using Hefesto.SII.Lce.Extruturas.Diario;
using Hefesto.SII.Lce.Extruturas.Diccionario;
using Hefesto.SII.Lce.Extruturas.Mayor;
using Hefesto.SII.Lce.Instancias;
using System.Text;
using System.Xml.Linq;            

internal class Program
{

    ///
    /// Punto de entrada del programa.
    static async Task Main(string[] args)
    {
        ///
        /// Generar el libro diccionario obligatorio.
        GenerarEnviolibroDiccionarioOblitatorio();
    }

	/// <summary>
	/// Crear el envio de libro diccionario obligatorio.
	/// </summary>
	public static void GenerarEnviolibroDiccionarioOblitatorio()
	{
		///
		/// crear el libro SiiLceEnvioLibros
		var bookdiccionarioOblig = HefestoLceElements.CrearSiiLceEnvioOblig()

			///
			/// Configuracion del libro
			.SiiConfiguracion(cfg =>
			{

				///
				/// Datos para la firma del documento.
				cfg.SiiAutenticacion(auth =>
				{
					auth.Basse64 = "MIIPNgIBAzCCDvIGCSqGSIb3DQEHAaCCDuMEgg7fMIIO2zCCBgwGCSq==";
					auth.Password = "1234";
				});

				///
				/// Datos del certificado del SII.
				cfg.SIILceCoCertificado(auth =>
				{
					auth.Load = "CAL_99999999-9_C_2_1.xml";
				});

			})


			///
			/// Documento envio obligatorio.
			.SiiDocumentoEnvioOblig(doc =>
			{

				doc.SiiCaratula(car =>
				{
					car.RutEnvia = "88888888-8";
					car.RutContribuyente = "99999999-9";
				});

				doc.SIILceCoCierre(cierre =>
				{
					///
					/// agregar libro diario res.
					cierre.SIIDiccionario( GenerarEnvioLibroDiccionario() );

				});

			});

		///
		/// Despliega en consola el resultado de la exportación del libro.
		var respExporta = bookdiccionarioOblig.ExportarLibro();
		Console.WriteLine($"EsCorrceto : {respExporta.Success}");
		Console.WriteLine($"Mensaje    : {respExporta.Message}");

		///
		/// Si es correcto, guarde el libro en un archivo XML.
		if (respExporta.Success)
			File.WriteAllText(
				"EnviosObligatorios\\EnvioLibroDiccionarioObligatorio.xml",
					respExporta.Data,
						Encoding.GetEncoding("ISO-8859-1"));

		///
		/// Crear el libro diccionario.
		HefLceDiccionario GenerarEnvioLibroDiccionario()
		{
			///
			/// Cree el libro.
			var libro = HefestoLceElements.CrearSiiLibroDiccionario();

			libro.SiiDocumentoDiccionario(doc =>
			{


				doc.SiiIdentificacion(ident =>
				{
					ident.RutContribuyente = "99999999-9";
					ident.PeriodoTributario = "2026";
					ident.Rectificatoria = 1;
				});

				doc.RutFirma = "88888888-8";

				///
				/// Agregar las cuentas.
				doc.SiiCuentas( GenerarCuentasDiccionario() );

			});

			///
			/// Regrese el libro diccionario.
			return libro;

		}

		///
		/// Crea las cuentas del balance.
		List GenerarCuentasDiccionario()
		{

			List cuentas = new();

			var cuenta = new HefCuentaDiccionario();
			cuenta.ClasificacionCuenta = ClasificacionCuenta.Activo;
			cuenta.CodigoCuenta = "1102001";
			cuenta.Glosa = "CAJA CHICA OFICINA SANTIAGO";
			cuenta.CodigoSII = "1220.010";
			cuentas.Add(cuenta);

			cuenta = new HefCuentaDiccionario();
			cuenta.ClasificacionCuenta = ClasificacionCuenta.Activo;
			cuenta.CodigoCuenta = "1131003";
			cuenta.Glosa = "CFONDOS POR RENDIR";
			cuenta.CodigoSII = "1220.070";
			cuentas.Add(cuenta);

			return cuentas;
		}

	}

}
