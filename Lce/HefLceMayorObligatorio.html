Hefesto LCE (Libro Mayor Obligatorio)

Libro Mayor Obligatorio del SII.
Registro contable que reúne y resume los movimientos de las cuentas de una empresa, permitiendo conocer sus saldos y respaldar la información contable declarada ante el Servicio de Impuestos Internos (SII).


Ejemplo de generación libro mayor obligatorio.



/// <summary>
/// Bibliotecas necsarias.
/// </summary>
using Hefesto.SII.Lce.Enumeradores;
using Hefesto.SII.Lce.Extruturas.Balance;
using Hefesto.SII.Lce.Extruturas.Diario;
using Hefesto.SII.Lce.Extruturas.Diccionario;
using Hefesto.SII.Lce.Extruturas.Mayor;
using Hefesto.SII.Lce.Instancias;
using System.Text;
using System.Xml.Linq;            

internal class Program
{

    ///
    /// Punto de entrada del programa.
    static async Task Main(string[] args)
    {
        ///
        /// Generar el libro mayor obligatorio.
        GenerarEnviolibroMayorOblitatorio();
    }

    /// <summary>
    /// Crear el envio de libro mayor obligatorio.
    /// </summary>
    public static void GenerarEnviolibroMayorOblitatorio()
    {

        ///
        /// crear el libro SiiLceEnvioLibros
        var bookMayorOblig = HefestoLceElements.CrearSiiLceEnvioOblig()

            ///
            /// Configuracion del libro
            .SiiConfiguracion(cfg =>
            {

                ///
                /// Datos para la firma del documento.
                cfg.SiiAutenticacion(auth =>
                {
                    auth.Basse64 = "MIIPNgIBAzCCDvIGCSqGSIb3DQEHAaCCDuMEgg7fMIIO2zCC==";
                    auth.Password = "1234";
                });

                ///
                /// Datos del certificado.
                cfg.SIILceCoCertificado(auth =>
                {

                    auth.Load = "CAL_99999999-9_C_2_1.xml";

                });

            })


            ///
            /// Documento envio obligatorio.
            .SiiDocumentoEnvioOblig(doc =>
            {

                doc.SiiCaratula(car =>
                {
                    car.RutEnvia = "88888888-8";
                    car.RutContribuyente = "99999999-9";
                });

                doc.SIILceCoCierre(cierre =>
                {
                    ///
                    /// agregar libro diario res.
                    cierre.SIIMayorRes( GenerarLibroMayorRes() );
                });

            });

        ///
        /// Despliega en consola el resultado de la exportación del libro.
        var respExporta = bookMayorOblig.ExportarLibro();
        Console.WriteLine($"EsCorrceto : {respExporta.Success}");
        Console.WriteLine($"Mensaje    : {respExporta.Message}");


        ///
        /// Si es correcto, guarde el libro en un archivo XML.
        if (respExporta.Success)
            File.WriteAllText(
                "EnviosObligatorios\\EnvioLibroMayorObligatorio.xml",
                    respExporta.Data,
                        Encoding.GetEncoding("ISO-8859-1"));


        ////
        //// Generación del libro diario res
        HefLceMayorRes GenerarLibroMayorRes()
        {

            ////
            //// Cree el libro.
            var libro = HefestoLceElements.CrearSiiLibroMayorRes();

            ////
            //// Complete el libro.
            libro.SiiDocumentoMayorRes(res => {

                res.SiiIdentificacion(ident => {

                    ident.RutContribuyente = "99999999-8";

                    ident.SiiPeriodoTributario(periodo => {

                        periodo.Inicial = "2026-01";
                        periodo.Final = "2026-01";

                    });

                });

                res.RutFirma = "88888888-8";

                ////
                //// Agregar las cuentas del libro mayor
                res.SiiCuentasMayor(CrearCuentasMayorRes());

            });

            ////
            //// Regrese el libro.
            return libro;

        }

        ////
        //// Crear las cuentas mayor res.
        List CrearCuentasMayorRes()
        {
            ////
            //// Lista de comprobantes diarios
            var cuentasMayorRes = new List();

            ////
            //// Registro #1
            var cuenta = new HefCuentaMayorRes
            {

                CodigoCuenta = "1.10",
                CantidadMovimientos = 1,
                Cierre = new HefCierreMayorRes
                {
                    MontosPeriodo = new HefMontosPeriodoMayorRes
                    {
                        Debe = 40658,
                        SaldoDeudor = 40658
                    },
                    MontosAcumulado = new HefMontosAcumuladoMayorRes
                    {
                        Debe = 40658,
                        SaldoDeudor = 40658
                    }
                }
            };

            cuentasMayorRes.Add(cuenta);

            ////
            //// Registro #2
            cuenta = new HefCuentaMayorRes
            {

                CodigoCuenta = "1.30",
                CantidadMovimientos = 3,
                Cierre = new HefCierreMayorRes
                {
                    MontosPeriodo = new HefMontosPeriodoMayorRes
                    {
                        Debe = 3134456,
                        Haber = 340658,
                        SaldoDeudor = 2793798
                    },
                    MontosAcumulado = new HefMontosAcumuladoMayorRes
                    {
                        Debe = 3134456,
                        Haber = 340658,
                        SaldoDeudor = 2793798
                    }
                }
            };

            cuentasMayorRes.Add(cuenta);

            ////
            //// Regrese el valor de retorno
            return cuentasMayorRes;

        }
    }
	
}
